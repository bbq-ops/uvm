     10      typedef struct packed{
     11          bit en;
     12          bit [1:0] prio;
     13          bit [2:0] len;
     14          bit [7:0] avail;
     15      } mcdf_reg_t;
     16  
     17      typedef enum {
     18          RW_LEN,
     19          RW_PRIO,
     20          RW_EN,
     21          R_AVAIL
     22      } mcdf_field_t;

     25      class mcdf_refmod;
     26          local string name;
     27          local virtual mcdf_intf intf;
     28          mcdf_reg_t regs[3];
     29          mailbox #(mon_data_t) in_mbs[3];
     30          mailbox #(fmt_trans) out_mbs[3];
     31          mailbox #(reg_trans) reg_mb;
     32          
     33          function new(string name = "mcdf_refmod");
     34              this.name = name;
     35              foreach(this.out_mbs[i]) this.out_mbs[i] = new();
     36          endfunction
     37          
     38          task run();
     39              fork
     40                  this.do_reset();
     41                  this.do_reg_update();
     42                  this.do_packed(0);
     43                  this.do_packed(1);
     44                  this.do_packed(2);
     45              join
     46          endtask
     47          
     48          task do_reg_update();
     49              reg_trans t;
     50              forever begin
     51                  this.reg_mb.get(t);
     52                  if(t.cmd == `WRITE && t.addr[7:4] == 0) begin
     53                      this.regs[t.addr[3:2]].en = t.data[0];
     54                      this.regs[t.addr[3:2]].len = t.data[5:3];
     55                      this.regs[t.addr[3:2]].prio = t.data[2:1];
     56                  end
     57                  else if(t.cmd == `READ && t.addr[7:4] == 1)
     58                      this.regs[t.addr[3:2]].avail = t.data[7:0];
     59              end
     60          endtask
     61          
     62          task do_packed(int id);
     63              fmt_trans ot;
     64              mon_data_t it;
     65              bit [2:0] len;
     66              
     67              forever begin
     68                  this.in_mbs[id].peek(it);
     69                  ot = new();
     70                  len = this.get_field_value(id,RW_LEN);
     71                  ot.length = (len >= 3) ? 6'd32 : (4 << len);
     72                  ot.ch_id = id;
     73                  ot.data = new[ot.length];
     74                  foreach(ot.data[i]) begin
     75                      in_mbs[id].get(it);
     76                      ot.data[i] = it.data;
     77                  end
     78                  out_mbs[id].put(ot);
     79              end  
     80          endtask
     81          
     82          function int get_field_value(int id,mcdf_field_t f);
     83              case(f)
     84                  RW_LEN : return(regs[id].len);
     85                  RW_PRIO : return(regs[id].prio);
     86                  RW_EN : return(regs[id].en);
     87                  R_AVAIL : return(regs[id].avail);
     88              endcase
     89          endfunction
     90          
     91          task do_reset();
     92              forever begin
     93                  @(negedge intf.rstn);
     94                  foreach(regs[i]) begin
     95                      regs[i].en = 'h1;
     96                      regs[i].prio = 'h3;
     97                      regs[i].len = 'h0;
     98                      regs[i].avail = 'h20;
     99                  end
     100              end
     101          endtask
     102          
     103          function void set_interface(virtual mcdf_intf intf);
     104              if(intf == null)
     105                  $error("interface have not been intantiated");
     106              else
     107                  this.intf = intf;
     108          endfunction
     109      endclass

     111      class mcdf_checker;
     112          local string name;
     113          local int err_count;
     114          local int total_count;
     115          local int chnl_count[3];
     116          local virtual chnl_intf chnl_vif[3];
     117          local virtual reg_intf reg_vif;
     118          local virtual mcdf_intf mcdf_vif;
     119          local virtual arb_intf arb_vif;
     120          mailbox #(mon_data_t) chnl_mbs[3];
     121          mailbox #(reg_trans) reg_mb;
     122          mailbox #(fmt_trans) fmt_mb;
     123          mailbox #(fmt_trans) exp_mbs[3];
     124          mcdf_refmod refmod;
     125          
     126          function new(string name = "mcdf_checker");
     127              this.name = name;
     128              this.refmod = new();
     129              foreach(this.chnl_mbs[i]) begin
     130                  this.chnl_mbs[i] = new();
     131                  this.refmod.in_mbs[i] = this.chnl_mbs[i];
     132                  this.exp_mbs[i] = this.refmod.out_mbs[i];
     133              end
     134              this.reg_mb = new();
     135              this.refmod.reg_mb = this.reg_mb;
     136              this.fmt_mb = new();
     137              
     138              this.err_count = 0;
     139              this.total_count = 0;
     140              foreach(this.chnl_count[i]) this.chnl_count[i] = 0;
     141          endfunction
     142          
     143          task run();
     144              fork
     145                  this.do_compare();
     146                  this.do_channel_disable_check(0);
     147                  this.do_channel_disable_check(1);
     148                  this.do_channel_disable_check(2);
     149                  this.do_arbiter_priority_check();
     150                  this.refmod.run();  
     151              join
     152          endtask
     153          
     154          function void set_interface(virtual chnl_intf chnl_vif[3],virtual arb_intf arb_vif,virtual mcdf_intf mcdf_vif);
     155              if(chnl_vif[0] == null || chnl_vif[1] == null || chnl_vif[2] == null)
     156                  $error("chnl_intf is not have been intantiated");
     157              else begin
     158                  this.chnl_vif = chnl_vif;
     159              end
     160              if(arb_vif == null) 
     161                  $error("arb_vif is not have been intantiated");
     162              else
     163                  this.arb_vif = arb_vif;
     164              if(mcdf_vif == null)
     165                  $error("mcdf_vif is not have been intantiated");
     166              else begin
     167                  this.mcdf_vif = mcdf_vif;
     168                  this.refmod.set_interface(mcdf_vif);
     169              end 
     170          endfunction
     171          
     172          task do_compare();
     173              fmt_trans expt,mont;
     174              bit cmp;
     175              forever begin
     176                  this.fmt_mb.get(mont);
     177                  this.exp_mbs[mont.ch_id].get(expt);
     178                  cmp = mont.compare(expt);
     179                  
     180                  this.total_count++;
     181                  this.chnl_count[mont.ch_id]++;
     182                  if(cmp == 0)begin
     183                      this.err_count++;
     184                      rpt_pkg::rpt_msg("[CMPFAIL]",$sformatf("%0t %0dth times comparing but failed! MCDF monitored output packet is different with reference model output",$time,this.total_count),
     185                      rpt_pkg::ERROR,
     186                      rpt_pkg::TOP,
     187                      rpt_pkg::LOG);
     188                  end
     189                  else begin
     190                      rpt_pkg::rpt_msg("[CMPSUCD]",$sformatf("%0t %0dth times comparing and succeeded! MCDF monitored output packet is the same with reference model output",$time,this.total_count),
     191                      rpt_pkg::INFO,
     192                      rpt_pkg::HIGH);
     193                  end    
     194              end
     195          endtask
     196          
     197          task do_channel_disable_check(int id);
     198             forever begin
     199              @(posedge this.mcdf_vif.clk iff (this.mcdf_vif.rstn) && (this.mcdf_vif.mon_ck.chnl_en[id] === 1'b0));
     200              if(this.chnl_vif[id].mon_ck.ch_ready === 1 && this.chnl_vif[id].mon_ck.ch_valid === 1)
     201                  rpt_pkg::rpt_msg("[CHKERR]",$sformatf("ERROR! %0t when channel disabled, ready signal raised when valid high",$time),
     202                  rpt_pkg::ERROR,
     203                  rpt_pkg::TOP);            
     204              end
     205          endtask
     206          
     207          task do_arbiter_priority_check();
     208              int id;
     209              forever begin
     210                  @(posedge this.arb_vif.clk iff(this.arb_vif.mon_ck.f2a_id_req === 1));
     211                  id = this.get_slave_id_with_prio();
     212                  
     213                  if(id >= 0) begin
     214                      @(posedge this.arb_vif.clk);
     215                      if(this.arb_vif.mon_ck.a2s_acks[id] !== 1)
     216                          rpt_pkg::rpt_msg("[CHKERR]",$sformatf("ERROR! %0t arbiter received f2a_id_req===1 and channel[%0d] raising request with high priority, but is not granted by arbiter",$time,id),
     217                          rpt_pkg::ERROR,
     218                          rpt_pkg::TOP);
     219                  end
     220              end
     221          endtask
     222          
     223          function int get_slave_id_with_prio();
     224              int id = -1;
     225              int prio = 999;
     226              foreach(this.arb_vif.mon_ck.slv_prios[i]) begin
     227                  if(this.arb_vif.mon_ck.slv_prios[i] < prio && this.arb_vif.mon_ck.slv_reqs[i] === 1) begin
     228                      id = i;
     229                      prio = this.arb_vif.mon_ck.slv_prios[i];
     230                  end
     231              end
     232              return id;
     233          endfunction
     234          
     235          function void do_report();
     236              string s;
     237              s = "\n---------------------\n";
     238              s = {s,"CHECKER SUMMARY \n"};
     239              s = {s,$sformatf("total comparison count: %0d \n",this.total_count)};
     240              foreach(this.chnl_count[i]) s = {s,$sformatf("channel[%0d] comparison count:%0d \n",i,this.chnl_count[i])};
     241              s = {s,$sformatf("total error count:%0d \n",this.err_count)};
     242              foreach(this.chnl_mbs[i]) begin
     243                  if(this.chnl_mbs[i].num() != 0)
     244                   s = {s,$sformatf("WARNING::chnl_mbs[%0d] is not empty! size = %0d \n",i,this.chnl_mbs[i].num())};
     245              end
     246              if(this.fmt_mb.num() != 0)
     247                  s = {s, $sformatf("WARNING:: fmt_mb is not empty! size = %0d \n", this.fmt_mb.num())}; 
     248              s = {s, "---------------------------------------------------------------\n"};
     249              rpt_pkg::rpt_msg($sformatf("[%s]",this.name), s, rpt_pkg::INFO, rpt_pkg::TOP);            
     250          endfunction
     251      endclass

     439          function new(string name = "mcdf_coverage");
     440              this.name = name;
     441              this.cg_mcdf_reg_write_read = new();
     442              this.cg_mcdf_reg_illegal_access = new();
     443              this.cg_channel_disable = new();
     444              this.cg_arbiter_priority = new();
     445              this.cg_formatter_grant = new();
     446              this.cg_formatter_length = new();
     447          endfunction
     448          
     449          task run();
     450              fork
     451                  this.do_reg_sample();
     452                  this.do_channel_sample();
     453                  this.do_arbiter_sample();
     454                  this.do_formatter_sample();
     455              join
     456          endtask
     457          
     458          task do_reg_sample();
     459              forever begin
     460                 @(posedge reg_vif.clk iff reg_vif.rstn);
     461                 this.cg_mcdf_reg_write_read.sample();
     462                 this.cg_mcdf_reg_illegal_access.sample();
     463              end
     464          endtask
     465          
     466          task do_channel_sample();
     467              forever begin
     468                  @(posedge mcdf_vif.clk iff mcdf_vif.rstn);
     469                  if(chnl_vifs[0].mon_ck.ch_valid===1 || 
     470                  chnl_vifs[1].mon_ck.ch_valid === 1 || 
     471                  chnl_vifs[2].mon_ck.ch_valid === 1)
     472                      this.cg_channel_disable.sample();
     473              end
     474          endtask
     475          
     476          task do_arbiter_sample();
     477              forever begin
     478                  @(posedge arb_vif.clk iff arb_vif.rstn);
     479                  if(arb_vif.slv_reqs[0] !== 0 || arb_vif.slv_reqs[1] !== 0 || arb_vif.slv_reqs[2] !== 0)
     480                      this.cg_arbiter_priority.sample();
     481              end
     482          endtask
     483          
     484          task do_formatter_sample();
     485              fork
     486                  forever begin
     487                      @(posedge fmt_vif.clk iff fmt_vif.rstn);
     488                      if(fmt_vif.mon_ck.fmt_req === 1)
     489                          this.cg_formatter_length.sample();
     490                  end
     491                  
     492                  forever begin
     493                      @(posedge fmt_vif.mon_ck.fmt_req);
     494                      this.delay_req_to_grant = 0;
     495                      forever begin
     496                          if(fmt_vif.fmt_grant === 1) begin
     497                              this.cg_formatter_grant.sample();
     498                              break;
     499                          end
     500                          else begin
     501                              @(posedge fmt_vif.clk);
     502                              this.delay_req_to_grant++;
     503                          end
     504                      end
     505                  end
     506              join
     507          endtask
     508
